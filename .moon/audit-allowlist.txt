# Paths the task audits ignore, shared by bin/audit-task-io.sh (writes) and
# bin/audit-task-reads.sh (reads). One path prefix per line.
#
# Only a tool's own derived cache belongs here. A real build artifact must be
# declared as an output, and a real source file must be declared as an input,
# or a cache hit serves stale content.
#
# A tool reading and writing its own cache is legitimately not a declared input
# or output: the cache is derived from the sources that ARE declared, and these
# tools all key their entries on content. Each entry still needs a reason,
# because the alternative is often to declare the path and let moon cache it,
# which can be the better trade.

# shadow-cljs incremental compilation state, ~146 MB. Declaring it would let a
# cold cljs build restore rather than recompile (24s today), at the cost of
# 146 MB per hash in the CAS. Worth revisiting.
.shadow-cljs/

# eslint's own cache. Declaring this one is probably worth it: a cold eslint run
# is 234.8s against 7.3s warm, so restoring it from the shared CAS would be the
# single largest win available here.
.eslintcache

# stylelint's cache, same shape as eslint's but the run is 8.2s not 235s.
.stylelintcache

# tsc incremental state. Lives under node_modules, which is already excluded,
# but listed so the reasoning is recorded.
node_modules/.cache/.tsbuildinfo

# Clojure CLI classpath cache, rebuilt from deps.edn which IS declared.
.cpcache/

# babel transform cache, rebuilt from the sources which ARE declared.
.babel_cache/

# clj-kondo analysis cache.
.clj-kondo/.cache/

# Generated by bun install via the prepare hook. The deps task declares
# .husky/_/** as an output, so writes are accounted for; reads are not inputs.
.husky/
